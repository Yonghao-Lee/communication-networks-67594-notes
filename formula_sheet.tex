\weekbar{01}{Lecture 1: Introduction \textperiodcentered\ Tutorial 1: Probability}

\begin{multicols}{2}

\section{Delay}
\begin{itemize}
  \item $d_{\text{nodal}}=d_{\text{proc}}+d_{\text{queue}}+d_{\text{trans}}+d_{\text{prop}}$: the four delays at each hop.
  \item $d_{\text{trans}}=L/R$: push $L$ bits onto a link of $R$ bit/s. Depends on size and rate, \hl{not distance}.
  \item $d_{\text{prop}}=d/s$: one bit travels distance $d$ at speed $s$ ($\approx2\times10^8$ m/s in fibre $=\frac23c$). Depends on distance, \hl{not size}.
  \item Speed-of-light bound: Jerusalem--NY $9164/3\times10^5\approx30.6$\,ms; in practice $\ge70$\,ms one way.
  \item CPU cycles wasted per RTT $=f_{\text{CPU}}\times\text{RTT}$ (3\,GHz $\times$ 140\,ms $=4.2\times10^8$).
\end{itemize}

\section{Circuit switching}
\begin{itemize}
  \item Phases: establish (costs $\ge1$ RTT) $\to$ transfer $\to$ teardown. Resources reserved; unused slots are lost.
  \item TDM/FDM with $n$ slots or bands: each circuit gets $R/n$. File time $=$ setup $+\,L/(R/n)$, independent of the number of links.
  \item Must reserve the peak $P$; utilization $=A/P$ ($A$ = average rate). Voice $\approx1/3$, bursty data $\le1\%$.
\end{itemize}

\section{Packet switching}
\begin{itemize}
  \item Store-and-forward, $N$ links, no queuing: one packet $N\frac LR$; $K$ packets back to back $(N+K-1)\frac LR$ (pipeline).
  \item Traffic intensity $La/R$ ($a$ = packets/s): must be $<1$, otherwise the queue grows without bound; near 1 the delay is huge.
  \item Statistical multiplexing: $N$ independent flows of mean $A$ need $\approx NA$, not $NP$. Spread: std $\propto\sqrt N$, std/mean $\propto1/\sqrt N$ (LLN). Fails if bursts are correlated.
  \item Users on a link: $X\sim\text{Bin}(n,q)$ active; overload probability $P(X>R/r)$ (35 users, $q=0.1$, 10 fit: $\approx0.0004$).
\end{itemize}

\section{Probability rules}
\begin{itemize}
  \item $P(A\mid B)=P(A\cap B)/P(B)$: $B$ known, ask about $A$.
  \item Independent: $P(A\cap B)=P(A)P(B)$ (\hl{multiply}). Disjoint: $P(A\cup B)=P(A)+P(B)$ (\hl{add}). Disjoint $\neq$ independent.
  \item Union bound: $P(A\cup B)\le P(A)+P(B)$; exactly: $P(A)+P(B)-P(A\cap B)$.
  \item Total probability: $P(D)=\sum_iP(D\mid B_i)P(B_i)$, $\{B_i\}$ a partition.
  \item Bayes: $P(B_j\mid D)=\dfrac{P(D\mid B_j)P(B_j)}{\sum_iP(D\mid B_i)P(B_i)}$.
  \item ``At least one of $n$ fails'' $=1-(1-p)^n$ (50 parts at 99\%: 39.5\%).
\end{itemize}

\section{Random variables}
\begin{itemize}
  \item $\E X=\sum x\,P(X=x)$; $\E(\sum a_iX_i)=\sum a_i\E X_i$ \hl{always} (even dependent). $\Var X=\E X^2-(\E X)^2$.
  \item Indicator: $\E[I_A]=P(A)$.
\end{itemize}
\begin{center}\footnotesize

\begin{itemize}
  \item Geometric: \emph{tries} until success, mean $1/p$; \emph{failures} before success, mean $(1-p)/p$.
  \item Bin$(n,p)\approx$ Pois$(np)$ when $n$ is large and $p$ small.
  \item Memoryless: $P(X>s+t\mid X>t)=P(X>s)$. Only Exp (continuous) and Geo (discrete).
\end{itemize}

\section{Poisson process (rate $\lambda$)}
\begin{itemize}
  \item Independent increments; count in any interval of length $t$ is Pois$(\lambda t)$; no bunching.
  \item Inter-arrival times $X_i$ (time \emph{since the previous} arrival) are i.i.d.\ Exp$(\lambda)$; epochs $S_i=X_1+\dots+X_i$.
  \item Exactly one arrival in $T$: $\lambda T e^{-\lambda T}$, maximal ($=1/e$) at $\lambda T=1$ (this is slotted ALOHA, week 2).
\end{itemize}

\section{Worked patterns}
\begin{itemize}
  \item Chain of $n$ links, each fails w.p.\ $p$: reach D w.p.\ $(1-p)^n$; end-to-end resend: $\E[\text{tries}]=1/(1-p)^n$; hop-by-hop: $1/(1-p)$ per hop, $n/(1-p)$ in total.
  \item Good/bad packet survival $e^{-t}$ / $e^{-1000t}$: $P(\text{surv})=pe^{-t}+(1-p)e^{-1000t}$; $P(\text{good}\mid\text{surv})$ by Bayes.
\end{itemize}

\begin{multicols}{2}

\section{Layers}
\begin{itemize}
  \item 5 layers (bottom up): Physical (bits) / Link (frames, local MAC addr.) / Network (packets, global addr., routing, \hl{only IP}) / Transport (segments, processes; TCP, UDP) / Application (messages).
  \item Routers: layers 1--3. Hosts: all 5.
  \item OSI: 7 layers, adding Session (5) and Presentation (6). In the Internet, OSI 5--7 are all ``application''.
  \item Layer = service (what) + interface (how to access) + protocol (how peers talk). Protocol = syntax (format, order) + semantics (meaning, actions).
  \item Encapsulation: each layer adds a header. TCP 20 + IP 20 + Ethernet 14 = \hl{54 B} of headers (+4 B CRC trailer, +8 B preamble on the wire). Useful fraction $=D/(D+54)$.
  \item Narrow waist: $A$ apps, $L$ links need $A+L$ adaptations instead of $A\times L$.
\end{itemize}

\section{MAC: the basics}
\begin{itemize}
  \item Wish list, rate $R$: one node alone $\to R$; $M$ busy $\to R/M$ each; decentralized; no clock sync; simple.
  \item TDMA/FDMA, $N$ stations: each gets $R/N$ \emph{always}, even when alone. No collisions; idle slots are wasted.
  \item Goodput (efficiency) = fraction of time with a \emph{successful} transmission. Throughput = fraction with \emph{any} transmission (in [KR]: a rate in bit/s). Useful rate $=\eta B$.
  \item $T=L/B$: slot = time to send one frame of $L$ bits at bandwidth $B$.
\end{itemize}

\section{ALOHA, binomial ($n$ nodes, each sends w.p.\ $p$)}
\begin{itemize}
  \item \textbf{Slotted}: one given node succeeds $p(1-p)^{n-1}$; some node succeeds (disjoint, add):
  \[ \eta=np(1-p)^{n-1},\qquad p^*=\tfrac1n, \]
  \[ \eta^*=(1-\tfrac1n)^{n-1}\to\tfrac1e\approx0.37. \]
  \item \textbf{Pure}: vulnerable window $2T$, each other node must skip 2 slots:
  \[ \eta=np(1-p)^{2(n-1)},\quad p^*=\tfrac1{2n-1},\quad \eta^*\to\tfrac1{2e}\approx0.18. \]
  \item $(1-\frac1N)^N\to\frac1e$ and $(1-\frac1N)^{N-1}\to\frac1e$.
  \item Slot types at $p=1/n$, many nodes: 37\% success, 37\% empty, 26\% collision.
  \item Finite-$n$ optimum is above the limit: slotted $n{=}2$: 0.50, $n{=}5$: 0.41; pure $n{=}2$: 0.296, $n{=}5$: 0.216.
\end{itemize}

\section{ALOHA, Poisson ($G$ attempts per slot)}
\begin{itemize}
  \item Slotted: $\eta=Ge^{-G}$, max $1/e$ at $G=1$. Empty slot $e^{-G}$; collision $1-e^{-G}-Ge^{-G}$.
  \item Pure: $\eta=Ge^{-2G}$, max $1/(2e)$ at $G=\frac12$.
  \item Binomial with $p=G/n$, $n\to\infty$ gives Poisson. If $\lambda$ is per second, $G=\lambda T$.
\end{itemize}

\section{Goodput: two views}
\begin{itemize}
  \item One slot: $\eta=\E[T_{\text{suc}}]/T=P_{\text{suc}}$.
  \item Wait for success: slots until success $\sim$ Geo$(P_{\text{suc}})$, so $\eta=T/(T/P_{\text{suc}})=P_{\text{suc}}$. Time between successes $=T/P_{\text{suc}}$.
\end{itemize}

\section{CSMA / CSMA/CD (Ethernet)}
\begin{itemize}
  \item CSMA: listen first. Collisions still happen because of the propagation delay. Plain CSMA wastes the whole frame.
  \item CD: abort when a collision is heard. Worst-case detection $\approx2\,\PROP$, so
  \[ \TRANSP\ge2\,\PROP\ \Longleftrightarrow\ L_{\min}=2\,\PROP\cdot R=\tfrac{2dR}{s}. \]
  \item Slot $=2\,\PROP$; won w.p.\ $\alpha=Np(1-p)^{N-1}$; wasted slots $A=\frac{1-\alpha}{\alpha}$ ($\alpha=0.4\Rightarrow A=1.5$).
  \item $\eta=\dfrac{\TRANSP}{\TRANSP+A\cdot2\,\PROP}=\dfrac1{1+3a}$ (simple); $\approx\dfrac1{1+5a}$ (precise), $a=\dfrac{\PROP}{\TRANSP}=\dfrac{t_{\text{prop}}}{t_{\text{trans}}}$.
  \item Backoff after the $m$-th collision: $K\in\{0,\dots,2^{\min(m,10)}-1\}$, wait $K\cdot512$ bit times. Bit time $=1/R$ (0.1\,$\mu$s at 10\,Mbps; $512\to51.2\,\mu$s). Two nodes collide again w.p.\ $1/2^m$. Jam = 48 bits.
\end{itemize}

\section{Addresses, WiFi}
\begin{itemize}
  \item MAC address: 48 bits, flat, portable; IEEE fixes 24 bits, the vendor 24 ($2^{24}$ per block). IP: hierarchical, not portable.
  \item 802.11: no CD (own signal drowns others; hidden terminal; fading) $\Rightarrow$ CSMA/CA: wait DIFS; if busy, random backoff that counts down only while idle; receiver ACKs after SIFS; no ACK $\Rightarrow$ larger backoff.
  \item RTS/CTS: short RTS (may collide, cheaply); the CTS is heard by everyone, including hidden nodes $\Rightarrow$ no data-frame collisions.
\end{itemize}

\section{Exam patterns (recitation)}
\begin{itemize}
  \item \emph{One specific} node succeeds: $p\cdot(\text{others silent})$, no factor $n$. \emph{Any} node (network goodput): multiply by $n$, i.e.\ $\binom n1$.
  \item Choice of network or case: law of total probability, $p\,P_1+(1-p)P_2$.
  \item Pure node vs.\ slotted node: each pure node must skip \emph{every} own slot that overlaps. Vulnerable window $=$ own length $+$ other's length.
  \item Q1: C one shot $=q[p(1-q)^2+(1-p)(1-q)^6]$; upper goodput $=p\cdot3q(1-q)^2+(1-p)\cdot2q(1-q)$.
  \item Q2 ($n$ pure nodes with slot $T$, $n$ slotted with slot $2T$): slotted node $p(1-p)^{n-1}(1-p)^{3n}$; pure node $\frac12p(1-p)^{2(n-1)}[(1-p)^n+(1-p)^{2n}]$.
  \item Q2c (all synchronized): with $s=np(1-p)^{n-1}$, $z=(1-p)^n$: $\eta=sz^2+sz=sz(1+z)$. Linearity of expectation avoids case splits.
\end{itemize}
